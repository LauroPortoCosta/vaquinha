procedure Tva_grupos.grava;
var
cc:string;
xx:string;
Begin
  xx:= '0';
  va_05_dm.DM.RESTClient_grupo.BaseURL:='https://laurolivro-606860-default-rtdb.firebaseio.com/tab_grupo/reg_grupo/'+xx+'.json?auth=gGrjaEZca7Dld3kCnSd7oDeQ8Xi8xJkcuvqCkrBg';
  va_05_dm.DM.RESTClient_grupo.Method := TRESTRequestMethod.rmPATCH;
  va_05_dm.DM.RESTClient_grupo.Params.Clear;
  va_05_dm.DM.RESTClient_grupo.Params.AddItem;
  va_05_dm.DM.RESTClient_grupo.Items[0].name:= 'param';
  cc:= '{"status":"n"}';
  va_05_dm.DM.RESTClient_grupo.Params.Items[0].value :=cc;
  va_05_dm.DM.RESTClient_grupo.Params.Items[0].ContentType := ctAPPLICATION_JSON;
  va_05_dm.DM.RESTClient_grupo.Params.Items[0].Kind := TRESTRequestParameterKind.pkGETorPOST;
  va_05_dm.DM.RESTClient_grupo.Execute;
  showmessage('salvo com sucesso');
